function renderActiveSeason() { let select = document.getElementById('yearSelect'); if (!select || select.options.length === 0) return; let selectedYear = parseInt(select.value); activeContests = rawContestsData.map(c => ({ ...c, scores: {...c.scores}, roundScores: {...c.roundScores}, details: [...c.details] })).filter(c => c.year === selectedYear); activeContests.sort((a, b) => a.date.getTime() - b.date.getTime()); document.getElementById('loadedCount').innerText = `(${activeContests.length} concours en ${selectedYear})`; if (activeContests.length === 0) { document.getElementById('tabContentsContainer').innerHTML = `

Aucun concours enregistré pour la saison ${selectedYear}.

`; return; } let seasonPilots = new Set(); activeContests.forEach(c => { Object.keys(c.scores).forEach(p => seasonPilots.add(p)); }); const K_ROUND = 24; let pilotEloGlobal = {}; seasonPilots.forEach(p => pilotEloGlobal[p] = 1200); // Boucle de convergence iterative sur la saison for (let iter = 0; iter < 15; iter++) { // 1. Calcul de la force moyenne des concours et de l'IDC activeContests.forEach(c => { let participants = Object.keys(c.scores); let sumElo = participants.reduce((acc, p) => acc + (pilotEloGlobal[p] || 1200), 0); c.contestElo = participants.length > 0 ? (sumElo / participants.length) : 1200; }); let totalContestElos = activeContests.reduce((acc, c) => acc + c.contestElo, 0); let meanNationalElo = activeContests.length > 0 ? (totalContestElos / activeContests.length) : 1200; // 2. Calcul et application sequentielle concours par concours activeContests.forEach(c => { c.idc = meanNationalElo > 0 ? (c.contestElo / meanNationalElo) : 1; c.quality = calculateContestQuality(c); let deltaElos = {}; seasonPilots.forEach(p => deltaElos[p] = 0); let rNames = Object.keys(c.roundScores || {}); if (rNames.length > 0) { rNames.forEach(rName => { let rScores = c.roundScores[rName]; let participants = Object.keys(rScores); let N = participants.length; if (N < 2) return; for (let i = 0; i < N; i++) { let pA = participants[i]; let sA = rScores[pA]; let rA = pilotEloGlobal[pA] || 1200; for (let j = 0; j < N; j++) { if (i === j) continue; let pB = participants[j]; let sB = rScores[pB]; let rB = pilotEloGlobal[pB] || 1200; let eA = 1 / (1 + Math.pow(10, (rB - rA) / 400)); let sVal = (sA > sB) ? 1 : ((sA === sB) ? 0.5 : 0); deltaElos[pA] += (K_ROUND * 0.1 / (N - 1)) * (sVal - eA); } } }); } else { let participants = Object.keys(c.scores); participants.sort((a, b) => c.scores[b] - c.scores[a]); let N = participants.length; if (N >= 2) { let K_eff = (300 * c.idc) / (N - 1); for (let i = 0; i < N; i++) { for (let j = i + 1; j < N; j++) { let pA = participants[i]; let pB = participants[j]; let rA = pilotEloGlobal[pA] || 1200; let rB = pilotEloGlobal[pB] || 1200; let eA = 1 / (1 + Math.pow(10, (rB - rA) / 400)); let sA = (c.scores[pA] === c.scores[pB]) ? 0.5 : 1; deltaElos[pA] += K_eff * (sA - eA) * 0.1; deltaElos[pB] += K_eff * ((1 - sA) - (1 - eA)) * 0.1; } } } } // Mise a jour immediate du rating pour le concours suivant seasonPilots.forEach(p => { pilotEloGlobal[p] = Math.max(800, (pilotEloGlobal[p] || 1200) + (deltaElos[p] || 0)); }); // Enregistrement des cotes post-concours a la derniere iteration if (iter === 14) { c.postContestElos = { ...pilotEloGlobal }; } }); } // Mise a jour des objets bruts activeContests.forEach(c => { let rawObj = rawContestsData.find(r => r.name === c.name && r.date.getTime() === c.date.getTime()); if (rawObj) { rawObj.idc = c.idc; rawObj.contestElo = c.contestElo; rawObj.quality = c.quality; } }); // Construction des classements de saison et historiques individuels currentSeasonRankings = []; let pilotsArr = Array.from(seasonPilots); pilotsArr.forEach(pilot => { let contestCount = 0; let sumPercent = 0; let history = []; let allScores = []; activeContests.forEach(c => { if (c.scores[pilot] !== undefined) { contestCount++; let pctScore = c.scores[pilot]; sumPercent += pctScore; allScores.push(pctScore); let eloAfter = Math.round(c.postContestElos && c.postContestElos[pilot] !== undefined ? c.postContestElos[pilot] : 1200); history.push({ contestName: c.name, date: c.date.toLocaleDateString(), raw: Math.round(pctScore * 10), percent: pctScore, eloAfter: eloAfter }); } }); // Règlement FFAM / FAI : conservation des 4 meilleurs résultats let best4Scores = [...allScores].sort((a, b) => b - a).slice(0, 4); let sumBest4Percent = best4Scores.reduce((a, b) => a + b, 0); let avgBest4Percent = best4Scores.length > 0 ? (sumBest4Percent / best4Scores.length) : 0; let finalEloVal = Math.round(pilotEloGlobal[pilot] || 1200); let avgPercent = contestCount > 0 ? (sumPercent / contestCount) : 0; let isOfficial = contestCount >= MIN_CONTESTS_REQUIRED; currentSeasonRankings.push({ pilot, cote: finalEloVal, elo: finalEloVal, sumPercent: sumBest4Percent, sumTotalPercent: sumPercent, avgPercent: avgBest4Percent, avgTotalPercent: avgPercent, contestCount, totalRounds: history.length, isOfficial }); currentSeasonHistories[pilot] = history; }); buildUI(); }
